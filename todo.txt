// JavaScript formatting functions
const formatters = {
    plain_text: (value) => value,
    
    google_maps_link: (value, businessName, suburb) => {
        const query = `${businessName}, ${value}, ${suburb}`;
        return `<a href="https://www.google.com/maps/search/?api=1&query=${encodeURIComponent(query)}" target="_blank">${value}</a>`;
    },
    
    tel_link: (value) => {
        const cleanPhone = value.replace(/[^\d+]/g, '');
        return `<a href="tel:${cleanPhone}">${value}</a>`;
    },
    
    instagram_link: (value) => {
        const handle = value.replace(/^@/, '');
        return `<a href="https://instagram.com/${handle}" target="_blank">@${handle}</a>`;
    },
    
    external_link: (value) => {
        const url = value.startsWith('http') ? value : `https://${value}`;
        return `<a href="${url}" target="_blank" rel="noopener">${value}</a>`;
    }
};

// Dynamic card rendering function
function renderBusinessCard(business, fieldMappings) {
    const card = document.createElement('article');
    card.className = 'card';
    
    // Sort fields by display order
    const sortedFields = fieldMappings.sort((a, b) => a.display_order - b.display_order);
    
    sortedFields.forEach(field => {
        const value = business[field.csv_column];
        if (!value) return;
        
        const element = document.createElement('div');
        element.className = field.css_class;
        
        // Apply appropriate formatting
        const formattedValue = formatters[field.field_type](value, business.name, business.suburb);
        element.innerHTML = formattedValue;
        
        card.appendChild(element);
    });
    
    return card;
}

// Database schema for field mappings
CREATE TABLE wp_blf_field_mappings (

    csv_column VARCHAR(100),
    field_type VARCHAR(50),
    display_label VARCHAR(100),
    display_order INT,
    is_searchable BOOLEAN DEFAULT FALSE,
    formatting_options JSON,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

Step 4: Smart Auto-Detection
function auto_detect_field_type($column_name, $sample_data) {
    $column_lower = strtolower($column_name);
    
    // Smart detection based on column names
    if (strpos($column_lower, 'name') !== false) return 'name';
    if (strpos($column_lower, 'address') !== false || strpos($column_lower, 'location') !== false) return 'address';
    if (strpos($column_lower, 'phone') !== false || strpos($column_lower, 'mobile') !== false) return 'phone';
    if (strpos($column_lower, 'instagram') !== false || strpos($column_lower, 'insta') !== false) return 'instagram';
    if (strpos($column_lower, 'website') !== false || strpos($column_lower, 'url') !== false) return 'website';
    
    // Pattern detection in sample data
    if (preg_match('/^@\w+/', $sample_data)) return 'instagram';
    if (preg_match('/^\+?[\d\s\-\(\)]+$/', $sample_data)) return 'phone';
    if (filter_var($sample_data, FILTER_VALIDATE_URL)) return 'website';
    
    return 'plain_text'; // Default fallback
}
